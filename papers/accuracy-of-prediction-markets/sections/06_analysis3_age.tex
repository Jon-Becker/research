\subsection{Main Result: The 5.4$\times$ Age Effect}
\label{subsec:a3_finding}

% Bar chart by age cohort
\begin{figure}[h]
\centering
\begin{tikzpicture}
  \begin{axis}[
    title={Calibration Accuracy by Market Age},
    xlabel={Market Duration},
    ylabel={Mean Absolute Error (\%)},
    ybar,
    bar width=0.65cm,
    width=0.9\textwidth,
    height=0.6\textwidth,
    xtick={1,2,3,4,5},
    xticklabels={$<$7d, 7-30d, 30-90d, 90-365d, $365+$d},
    x tick label style={rotate=45, anchor=east}
  ]
    \addplot[fill=darkred!70] coordinates {
      (1, 22.56)
      (2, 9.57)
      (3, 11.08)
      (4, 8.06)
      (5, 4.18)
    };
  \end{axis}
\end{tikzpicture}
\caption{Market calibration improves monotonically over market lifetime. Very-short markets (resolving within 7 days) show 22.56\% MAE; very-long markets (365+ days) show 4.18\% MAE, representing a 5.4$\times$ improvement. Note: the non-monotonic pattern in medium-duration markets (30--90d showing slightly higher MAE than 7--30d) suggests confounding with market characteristics or information arrival timing.}
\label{fig:age_effect}
\end{figure}

\subsection{The Confound: Age vs. Volume Accumulation}
\label{subsec:a3_confound}

% Age-volume correlation analysis with scatter plot
\begin{figure}[h]
\centering
\begin{tikzpicture}
  \begin{axis}[
    title={Volume Accumulation Over Market Lifetime},
    xlabel={Market Age (days, log scale)},
    ylabel={Median Trading Volume (log scale)},
    xmode=log,
    ymode=log,
    width=0.8\textwidth,
    height=0.6\textwidth,
    grid=major
  ]
    \addplot[only marks, mark=o, mark size=4pt, color=darkgreen, opacity=0.6] coordinates {
      (1, 1)
      (7, 10)
      (30, 100)
      (90, 500)
      (365, 19187)
    };
    \addlegend{Median volume by age cohort}
  \end{axis}
\end{tikzpicture}
\caption{Strong positive correlation between market age and cumulative trading volume (Pearson $r = 0.85$). Older markets accumulate substantially more trades, confounding age effects with volume effects. Both mechanisms---information arrival and volume accumulation---are plausible drivers of improved calibration, but our observational design cannot separate them.}
\label{fig:age_volume_confound}
\end{figure}
