\subsection{Main Result: The 4.2$\times$ Volume Effect}
\label{subsec:a2_finding}

% Bar chart showing volume cohorts
\begin{figure}[h]
\centering
\begin{tikzpicture}
  \begin{axis}[
    title={Calibration Accuracy by Volume Cohort},
    xlabel={Volume Cohort},
    ylabel={Mean Absolute Error (\%)},
    ybar,
    bar width=0.7cm,
    width=0.9\textwidth,
    height=0.6\textwidth,
    xtick={1,2,3,4,5},
    xticklabels={Ultra-Thin, Thin, Medium, Liquid, Very Liquid},
    x tick label style={rotate=45, anchor=east},
    legend pos=upper right,
    error bars/.cd,
    y dir=both,
    y explicit
  ]
    \addplot[fill=darkblue!70, error bars={y=error}] coordinates {
      (1, 16.37) +- (0, 0.13)
      (2, 12.84) +- (0, 0.14)
      (3, 9.13) +- (0, 0.15)
      (4, 7.29) +- (0, 0.16)
      (5, 3.87) +- (0, 0.13)
    };
    \addlegend{MAE with 95\% CI}
  \end{axis}
\end{tikzpicture}
\caption{Calibration accuracy improves monotonically across volume cohorts. Ultra-thin markets (fewer than 100 trades) show 16.37\% MAE; very-liquid markets (more than 10,000 trades) show 3.87\% MAE, representing a 4.2$\times$ improvement.}
\label{fig:volume_effect}
\end{figure}
